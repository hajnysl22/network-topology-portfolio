CORE-R2
enable
configure terminal

hostname CORE-R2
ip domain-name lab.local
service password-encryption

crypto key generate rsa modulus 2048
ip ssh version 2

no ip http server
no ip http secure-server
no ip source-route
no ip finger
no ip bootp server
no service config

login block-for 120 attempts 3 within 60
login delay 2
login on-failure log
login on-success log

username admin privilege 15 secret AdminSecurePass123

aaa new-model

radius server SRV2_RADIUS
 address ipv4 192.168.20.20 auth-port 1812 acct-port 1813
 key RadiusKey123
exit

aaa group server radius RADIUS_GROUP
 server name SRV2_RADIUS
exit

aaa authentication login default local
aaa authorization exec default local

interface Loopback0
 description Management Loopback
 ip address 2.2.2.2 255.255.255.255
 no shutdown
exit

interface fa0/0
 description Connection to EDGE-R2
 ip address 10.0.0.6 255.255.255.252
 no shutdown
exit

interface fa1/0
 description Connection to DIST-R1
 ip address 10.0.1.13 255.255.255.252
 ip ospf network point-to-point
 ip ospf message-digest-key 1 md5 OspfSecurePass26
 no shutdown
exit

interface fa2/0
 description Connection to DIST-R2
 ip address 10.0.1.17 255.255.255.252
 ip ospf network point-to-point
 ip ospf message-digest-key 1 md5 OspfSecurePass26
 no shutdown
exit

interface fa3/0
 description Connection to DIST-R3
 ip address 10.0.1.21 255.255.255.252
 ip ospf network point-to-point
 ip ospf message-digest-key 1 md5 OspfSecurePass26
 no shutdown
exit

interface fa4/0
 description Connection to CORE-R1
 ip address 10.0.0.10 255.255.255.252
 ip ospf network point-to-point
 ip ospf message-digest-key 1 md5 OspfSecurePass26
 no shutdown
exit

interface range FastEthernet 1/0 , FastEthernet 2/0 , FastEthernet 3/0 , FastEthernet 4/0
 speed 100
 duplex full
exit

ip sla 1
 icmp-echo 203.0.113.5 source-interface fa0/0
 frequency 5
exit

ip sla schedule 1 life forever start-time now

track 1 ip sla 1 reachability

ip route 203.0.113.5 255.255.255.255 10.0.0.5
ip route 0.0.0.0 0.0.0.0 10.0.0.5 track 1

router ospf 1
 router-id 2.2.2.2
 network 2.2.2.2 0.0.0.0 area 0
 network 10.0.0.8 0.0.0.3 area 0
 network 10.0.1.12 0.0.0.3 area 0
 network 10.0.1.16 0.0.0.3 area 0
 network 10.0.1.20 0.0.0.3 area 0
 default-information originate metric 10 metric-type 1
 area 0 authentication message-digest
exit

ip access-list standard ACL_ADMIN_MGMT
 permit 192.168.99.0 0.0.0.255
 permit 192.168.20.0 0.0.0.255
 deny any log
exit

ip access-list extended ACL_COPP_MGMT
 permit tcp 192.168.99.0 0.0.0.255 any eq 22
 permit tcp 192.168.20.0 0.0.0.255 any eq 22
 permit ospf any any
 permit icmp any any echo
 permit icmp any any echo-reply
exit

class-map match-all CM_COPP_MGMT
 match access-group name ACL_COPP_MGMT
exit

policy-map PM_COPP
 class CM_COPP_MGMT
  police 1000000 conform-action transmit exceed-action transmit
 class class-default
  police 256000 conform-action transmit exceed-action drop
exit

control-plane
 service-policy input PM_COPP
exit

logging host 192.168.20.20
logging trap informational
logging source-interface Loopback0

snmp-server community MgtSecureRO_2026 RO ACL_ADMIN_MGMT
snmp-server trap-source Loopback0
snmp-server host 192.168.20.20 version 2c labpublic

line con 0
 login local
 exec-timeout 0 0
exit

line vty 0 4
 login authentication default
 authorization exec default
 transport input ssh
 exec-timeout 0 0
exit

line vty 5 15
 transport input none
 no exec
exit

line vty 0 15
 access-class ACL_ADMIN_MGMT in
exit
