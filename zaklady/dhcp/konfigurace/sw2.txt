enable
	configure terminal
	hostname SW2

	ip domain-name pokus.local
	crypto key generate rsa general-keys modulus 2048
	username admin secret pokus
	ip ssh version 2
	enable secret pokus
	service password-encryption

	ip dhcp snooping
	ip dhcp snooping vlan 10,20

	line vty 0 4
	login local
	transport input ssh

	spanning-tree mode rapid-pvst

	interface range FastEthernet 0/1 - 24
		spanning-tree portfast
		spanning-tree bpduguard enable 

	vlan 10
		name VLAN-10
	vlan 20
		name VLAN-20
	vlan 30
		name VLAN-30
	vlan 1000
		name Native-VLAN
	
	interface range FastEthernet 0/1 - 10
		switchport mode access
		switchport access vlan 10

	interface range FastEthernet 0/11 - 20
		switchport mode access
		switchport access vlan 20

	interface FastEthernet 0/24
		switchport mode access
		switchport access vlan 30
		ip dhcp snooping trust

	interface GigabitEthernet 0/1
		switchport mode trunk
		switchport trunk allowed vlan 10,20,30
		switchport trunk native vlan 1000
		switchport nonegotiate 
		ip dhcp snooping trust

	interface VLAN 30 
		ip address 192.168.30.20 255.255.255.0

	interface range fastEthernet 0/1-24
		switchport mode access
		switchport port-security
		switchport port-security mac-address sticky

	interface range fastEthernet 0/1-23
		ip dhcp snooping limit rate 4

	interface range fastEthernet 0/2-10 
		shutdown
	interface range fastEthernet 0/12-23 
		shutdown

	end
write
	show vlan 
	show ip interface brief
	show etherchannel summary
	show port-security address
	show ip dhcp snooping
