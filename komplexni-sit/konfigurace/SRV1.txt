sudo apt install isc-dhcp-server

    sudo nano /etc/default/isc-dhcp-server
    INTERFACESv4="eth0"

    sudo nano /etc/dhcp/dhcpd.conf
    option domain-name "lab.local";
    option domain-name-servers 192.168.20.10, 192.168.20.20;
    default-lease-time 86400;
    max-lease-time 604800;
    authoritative;

    failover peer "dhcp-failover" {
        primary;
        address 192.168.20.10;
        port 647;
        peer address 192.168.20.20;
        peer port 647;
        max-response-delay 30;
        max-unacked-updates 10;
        load balance max seconds 3;
        mclt 1800;
        split 128;
    }

    # VLAN 20 - Servers
    subnet 192.168.20.0 netmask 255.255.255.0 {
        option routers 192.168.20.1;
        pool {
            failover peer "dhcp-failover";
            range 192.168.20.50 192.168.20.99;
        }
    }

    # VLAN 10 - Users
    subnet 192.168.10.0 netmask 255.255.255.0 {
        option routers 192.168.10.1;
        pool {
            failover peer "dhcp-failover";
            range 192.168.10.50 192.168.10.200;
        }
    }

    # VLAN 30 - WiFi
    subnet 192.168.30.0 netmask 255.255.255.0 {
        option routers 192.168.30.1;
        pool {
            failover peer "dhcp-failover";
            range 192.168.30.50 192.168.30.200;
        }
    }

    # VLAN 99 - Management
    subnet 192.168.99.0 netmask 255.255.255.0 {
        option routers 192.168.99.1;
        pool {
            failover peer "dhcp-failover";
            range 192.168.99.50 192.168.99.100;
        }
    }

sudo apt install bind9

    sudo nano /etc/bind/named.conf.options
    options {
        directory "/var/cache/bind";

        // Allow queries from internal subnets
        allow-query { 127.0.0.1; 192.168.0.0/16; 10.99.99.0/24; 10.0.0.0/8; };
        recursion yes;

        // Upstream forwarders for public internet resolution
        forwarders {
            8.8.8.8;
            1.1.1.1;
        };

        dnssec-validation auto;
        listen-on-v6 { any; };
    };

    sudo nano /etc/bind/named.conf.local
    zone "lab.local" {
        type master;
        file "/etc/bind/db.lab.local";
        allow-transfer { 192.168.20.20; }; // Allow transfer to SRV2
    };

    zone "20.168.192.in-addr.arpa" {
        type master;
        file "/etc/bind/db.192.168.20";
        allow-transfer { 192.168.20.20; };
    };

    sudo nano /etc/bind/db.lab.local
    $TTL    604800
    @       IN      SOA     srv1.lab.local. admin.lab.local. (
                                1         ; Serial
                            604800         ; Refresh
                            86400         ; Retry
                            2419200         ; Expire
                            604800 )       ; Negative Cache TTL
    ; Name Servers
    @       IN      NS      srv1.lab.local.
    @       IN      NS      srv2.lab.local.

    ; Host Records
    srv1    IN      A       192.168.20.10
    srv2    IN      A       192.168.20.20
    fw      IN      A       192.168.20.1
    asav    IN      A       10.99.99.1
    ap      IN      A       192.168.99.20

    sudo nano /etc/bind/db.192.168.20
    $TTL    604800
    @       IN      SOA     srv1.lab.local. admin.lab.local. (

sudo apt install openssl

    sudo mkdir -p /etc/ssl/lab-ca/{certs,crl,newcerts,private}
    sudo chmod 700 /etc/ssl/lab-ca/private
    cd /etc/ssl/lab-ca

    sudo openssl req -x509 -new -nodes -keyout private/ca.key -sha256 -days 3650 -out certs/ca.crt
    -subj "/C=CZ/ST=Czechia/L=Lab/O=Lab Network/CN=Lab Root CA"

    sudo openssl req -new -nodes -keyout /etc/ssl/private/web.key -out /etc/ssl/web.csr
    -subj "/C=CZ/ST=Czechia/L=Lab/O=Lab Network/CN=srv1.lab.local"

    nano /etc/ssl/web.ext
    authorityKeyIdentifier=keyid,issuer
    basicConstraints=CA:FALSE
    keyUsage = digitalSignature, nonRepudiation, keyEncipherment, dataEncipherment
    subjectAltName = @alt_names

    [alt_names]
    DNS.1 = srv1.lab.local
    DNS.2 = srv1
    IP.1 = 192.168.20.10
    EOF

    sudo openssl x509 -req -in /etc/ssl/web.csr -CA certs/ca.crt -CAkey private/ca.key
    -CAcreateserial -out /etc/ssl/certs/web.crt -days 825 -sha256 -extfile /etc/ssl/web.ext
